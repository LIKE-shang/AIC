cmake_minimum_required(VERSION 3.0.2)
project(full_chain_vision_20261025_v6_1)
find_package(catkin REQUIRED COMPONENTS cv_bridge geometry_msgs image_transport mavros_msgs roscpp rospy sensor_msgs std_msgs)
find_package(Threads REQUIRED)
find_package(OpenSSL REQUIRED)
catkin_python_setup()
catkin_package()
include_directories(${catkin_INCLUDE_DIRS} ${OPENSSL_INCLUDE_DIR})
add_executable(full_chain_target_laser_v6_1 src/hover_target_laser.cpp)
set_target_properties(full_chain_target_laser_v6_1 PROPERTIES COMPILE_FLAGS "-std=c++11")
target_link_libraries(full_chain_target_laser_v6_1 ${catkin_LIBRARIES} ${CMAKE_DL_LIBS} ${CMAKE_THREAD_LIBS_INIT} ${OPENSSL_LIBRARIES})
catkin_install_python(PROGRAMS scripts/circle_full_chain_servo_v6_1.py scripts/record_full_chain_v6_1.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(TARGETS full_chain_target_laser_v6_1 RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY launch config DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
